DECLARE 
    v_street  VARCHAR2(255); 
    v_city_name  VARCHAR2(50); 
    v_payload    CLOB; 
    v_message    VARCHAR2(200); 
    v_occurredAt TIMESTAMP; 
    v_performedBy VARCHAR2(200); 
BEGIN 
/*  
    Trigger informuje system rodo o zmianach danych kontrahentów;  
    %author : Waldemar Sudoł  
    %date   : 2025-11-13  
    %changelog:  
        2020-11-13 Waldemar Sudoł[47015]  
        - Utworzenie triggera

 IF UPDATING AND :NEW.STATUS = 'ZM' AND :NEW.STATUS != :OLD.STATUS THEN 
     v_message    := 'AUMS.Integration.Messages.Subjects.SubjectAnonymized'; 
     v_occurredAt := :NEW.DT_UPDATED; 
     v_performedBy := :NEW.USER_UPDATED;

     v_message    := 'AUMS.Integration.Messages.Subjects.SubjectUpdated'; 
     v_occurredAt := :NEW.DT_UPDATED; 
     v_performedBy := :NEW.USER_UPDATED; 
 ELSE 
     v_message    := 'AUMS.Integration.Messages.Subjects.SubjectCreated'; 
     v_occurredAt := :NEW.DT_INSERTED; 
     v_performedBy := :NEW.USER_INSERTED; 
 END IF; 
 
    IF :NEW.KSE_USER_TYPE IS NOT NULL AND :NEW.NRW_ADRESU IS NOT NULL THEN 
  SELECT 
      aa.STREET, 
      ac.CITY_NAME 
  INTO v_street, v_city_name 
  FROM AD_ADDRESS aa 
  INNER JOIN AD_CITY ac ON ac.id_city = aa.id_city 
  WHERE aa.ID_ADDRESS = :NEW.NRW_ADRESU; 
 
  v_payload := TO_CLOB(JSON_OBJECT( 
               'SubjectId'        VALUE :NEW.NRW_KONTRAHENTA, 
               'KseUserType'      VALUE :NEW.KSE_USER_TYPE, 
               'FirstName'        VALUE :NEW.IMIE, 
               'LastName'         VALUE :NEW.NAZWISKO,

      p_message_type => v_message, 
      p_payload      => v_payload

    END IF; 
END;

  CREATE OR REPLACE EDITIONABLE TRIGGER "SKOK_PROD"."TRG_MAILING_ADDRESS_CHANGE_KK_KONTRAHENCI" 
AFTER UPDATE OF NRW_ADRESU_KORESP 
ON KK_KONTRAHENCI 
FOR EACH ROW 
DECLARE 
    v_dom               KK_ADRES_KORESP.DOM%TYPE; 
    v_lokal             KK_ADRES_KORESP.LOKAL%TYPE; 
    v_parcel            KK_ADRES_KORESP.PARCEL%TYPE; 
    v_opis1             KK_ADRES_KORESP.OPIS1%TYPE; 
    v_opis2             KK_ADRES_KORESP.OPIS2%TYPE; 
    v_korespAdresId     KK_ADRES_KORESP.NRW_ADRESU%TYPE; 
    v_tmpUlica          kk_adres_koresp.tmp_ulica%TYPE; 
    v_tmpMiasto         kk_adres_koresp.tmp_miasto%TYPE; 
    v_tmpKodPocztowy    kk_adres_koresp.tmp_kod%TYPE; 
    v_tmpKraj           kk_adres_koresp.tmp_kraj%TYPE; 
    v_count   INTEGER; 
BEGIN 
    IF :NEW.IS_BILLING_ACCOUNT = 'T' THEN 
        IF 
            (:OLD.NRW_ADRESU_KORESP IS NULL AND :NEW.NRW_ADRESU_KORESP IS NOT NULL) OR 
            (:OLD.NRW_ADRESU_KORESP IS NOT NULL AND :NEW.NRW_ADRESU_KORESP IS NULL) OR 
            (:OLD.NRW_ADRESU_KORESP <> :NEW.NRW_ADRESU_KORESP) 
        THEN 
            IF :NEW.NRW_ADRESU_KORESP IS NOT NULL THEN 
                -- sprawdzenie czy istnieje adres 
                SELECT COUNT(*) 
                INTO v_count 
                FROM kk_adres_koresp 
                WHERE nrw_adresu_koresp = :NEW.NRW_ADRESU_KORESP; 
  
                IF v_count > 0 THEN 
                    SELECT dom, lokal, parcel, opis1, opis2, nrw_adresu, tmp_ulica, tmp_miasto, tmp_kod, tmp_kraj 
                    INTO v_dom, v_lokal, v_parcel, v_opis1, v_opis2, v_korespAdresId, v_tmpUlica, v_tmpMiasto, v_tmpKodPocztowy, v_tmpKraj 
                    FROM kk_adres_koresp 
                    WHERE nrw_adresu_koresp = :NEW.NRW_ADRESU_KORESP; 
  
                    FRSP_SAVE_EVENT_MAILING_ADDRESS_UPDATE( 
                        pn_EntityId     => :NEW.NRW_KONTRAHENTA, 
                        pn_AdresId      => v_korespAdresId, 
                        ps_Dom          => v_dom, 
                        ps_Lokal        => v_lokal, 
                        ps_Parcel       => v_parcel, 
                        ps_Opis1        => v_opis1, 
                        ps_Opis2        => v_opis2, 
                        ps_TmpUlica     => v_tmpUlica, 
                        ps_TmpMiasto    => v_tmpMiasto, 
                        ps_TmpKodPoczta => v_tmpKodPocztowy, 
                        ps_TmpKraj      => v_tmpKraj 
                    ); 
                END IF; 
            END IF; 
        END IF; 
    END IF; 
END;

    Purpose: Publikacja eventu usunięcia adresu korespondencyjnego konta rozliczeniowego do 
             outboksa (AUMS.Integration.Messages.BillingAccounts.BillingAccountCorrespondenceAddressDeletedMessage)

        2026-07-16 Kacper Kozera [64499]

  v_payload CLOB;

  IF :NEW.IS_BILLING_ACCOUNT = 'T' 
     AND COALESCE(:OLD.NRW_ADRESU_KORESP, 0) > 0

      p_message_type => 'AUMS.Integration.Messages.BillingAccounts.BillingAccountCorrespondenceAddressDeletedMessage', 
      p_payload      => v_payload 
    ); 
  END IF;

  CREATE OR REPLACE EDITIONABLE TRIGGER "SKOK_PROD"."JPTR_DANEKONTR_IDJEDNBIZ" 
  BEFORE INSERT OR UPDATE ON JP_DANE_KONTR 
  FOR EACH ROW 
--------------------------------------------------------------------------------------- 
  -- %author : ZPB JAN.GRODECKI 
  -- %date   : 2020-11-10 
  -- Trigger dodaje dane do kolumny rejon 
 
  --------------------------------------------------------------------------------------- 
DECLARE 
  V CONSTANT VARCHAR2(10) := '@#$%^NH('; --zaślepka dla napisu 
  N CONSTANT NUMBER := -5206; --zaślepka dla number 
BEGIN 
    IF UPDATING THEN 
        IF (coalesce(:new.rejon, V) <> coalesce(:old.rejon, V))THEN 
             raise_application_error(-20001,'Próba wpisania wartości do kolumny rejon'); 
 
        ELSIF (coalesce(:new.id_jedn_biz, N) <> coalesce(:old.id_jedn_biz, N)) and (:new.id_jedn_biz is null)THEN 
              :new.rejon := null; 
 
        ELSIF (coalesce(:new.id_jedn_biz, N) <> coalesce(:old.id_jedn_biz, N))THEN 
             select symbol into :new.rejon from jp_jedn_biz where id_jedn_biz= :new.id_jedn_biz; 
         
        END IF; 
 
    END IF; 
 
    IF INSERTING THEN 
        IF :new.rejon is not null or :new.rejon <>'' THEN 
             raise_application_error(-20001,'Próba wpisania wartości do kolumny rejon'); 
 
        ELSIF :new.id_jedn_biz is not null then 
             select symbol into :new.rejon from jp_jedn_biz where id_jedn_biz= :new.id_jedn_biz; 
 
        END IF; 
         
    END IF; 
 
END JPTR_DANEKONTR_IDJEDNBIZ;

  CREATE OR REPLACE EDITIONABLE TRIGGER "SKOK_PROD"."JPTR_DANEKONTR_BILLACCOUNT" 
BEFORE UPDATE ON JP_DANE_KONTR 
FOR EACH ROW 
 
/* 
    Trigger zabezpiecza przed zmianą dane konta rozliczeniowego pochodzące z pierwszego nabywcy; 
    %author : PB-Team Lukasz Mainka 
    %date   : 2020-12-15 
    %changelog: 
        2020-11-26 Lukasz Mainka [NF-005316-2020] 
        - Utworzenie triggera 
        2021-01-11 Tomasz Siuchninski [R-000179-2021] 
        - Utworzenie wersji ORA 
        2024-03-29 Tomasz Pomianowski [NF-000393-2024] 
        - odświeżenie wersji Oracle dla ENEA i Tauron 
*/ 
 
DECLARE 
    n_cnt           number(10); 
 
BEGIN 
 
 IF cusf_is_billing_account(:OLD.nrw_kontrahenta) > 0 THEN 
 
        SELECT count(*) 
        INTO n_cnt 
        FROM kk_kontrahenci kk 
        WHERE kk.nrw_kontrahenta = :NEW.nrw_kontrahenta AND kk.status = 'AK' 
            AND NOT EXISTS(SELECT 1 FROM kk_billing_account_update_tmp WHERE id_customer = :OLD.nrw_kontrahenta); 
 
        IF n_cnt > 0 THEN 
 
            :NEW.typ_odb := :OLD.typ_odb; 
            :NEW.ekd := :OLD.ekd; 
            :NEW.branza := :OLD.branza; 
            :NEW.kapital_zakladowy := :OLD.kapital_zakladowy; 
            :NEW.kapital_wplacony := :OLD.kapital_wplacony; 
 
        END IF; 
 
    END IF; 
 
END JPTR_DANEKONTR_BILLACCOUNT;

  CREATE OR REPLACE EDITIONABLE TRIGGER "SKOK_PROD"."JPTR_UPDATE_BILLING_ACCOUNT" 
FOR INSERT OR update ON JP_DANE_KONTR 
COMPOUND TRIGGER 
 
/* 
    Trigger synchronizujący dane konta rozliczeniowego z jego pierwszym nabywcą 
    %author : OK-Team Waldemar Grzywiński 
    %date   : 2020-10-19 
    %changelog: 
        2020-10-19 Waldemar Grzywiński [NF-004283-2020] 
        - Trigger function created 
        2021-01-11 Tomasz Siuchniński [R-000179-2021] 
        - Utworzenie wersji dla ORA 
        2024-03-28 Tomasz Pomianowski [NF-000393-2024] 
        - odświeżenie wersji Oracle dla ENEA i Tauron 
*/ 
 
    TYPE t_nrw_tab is table of jp_dane_kontr.nrw_kontrahenta%type index by pls_integer; 
    vcNrwKontrahenta t_nrw_tab;  
    BEFORE STATEMENT IS 
    BEGIN 
        NULL; 
    END BEFORE STATEMENT; 
 
    BEFORE EACH ROW IS 
    BEGIN 
        NULL; 
    END BEFORE EACH ROW; 
 
    AFTER EACH ROW IS 
    BEGIN 
        vcNrwKontrahenta(vcNrwKontrahenta.count+1) := :new.nrw_kontrahenta; 
    END AFTER EACH ROW; 
 
    AFTER STATEMENT IS 
    BEGIN 
        FOR i in 1 .. coalesce(vcNrwKontrahenta.count,0) LOOP 
            kksp_update_billing_account(vcNrwKontrahenta(i), 2); 
        END LOOP; 
    END AFTER STATEMENT; 
 
END JPTR_UPDATE_BILLING_ACCOUNT;
